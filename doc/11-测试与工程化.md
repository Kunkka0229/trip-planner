```python
def make_fake_amap(fail: set[str] | None = None) -> AmapClient:
    """fail：让哪些工具失败，可选 {"poi", "weather"}"""
    fail = fail or set()

    def handler(request: httpx.Request) -> httpx.Response:
        if request.url.path.endswith("/weather/weatherInfo"):
            if "weather" in fail:
                return httpx.Response(200, json={"status": "0", "info": "DAILY_QUERY_OVER_LIMIT"})
            return httpx.Response(200, json=WEATHER_RESPONSE)
        if "poi" in fail:
            return httpx.Response(500)
        return httpx.Response(200, json=POI_RESPONSE)

    # 第 5 课预留的 transport 参数，就是为了这里：注入一个不联网的"假网络"
    # qps_limit=0：假网络不需要限速（第 9 课加的），否则每个测试都要白白等待
    return AmapClient("fake-key", transport=httpx.MockTransport(handler), qps_limit=0)


class FakeLLM(BaseChatModel):
    """按顺序返回预设的回复；回复是异常时就抛出，用来模拟超时

    继承 BaseChatModel，只需要实现 _generate：invoke、重试、回调这些通用逻辑都由父类提供，
    和 ChatOpenAI 走的是同一条调用链，所以代码里的 llm.invoke(...) 不用做任何修改
    """

    # BaseChatModel 本身是 Pydantic 模型，所以字段要这样声明（和 schemas.py 的写法一样）
    replies: list[str | Exception]
    calls: list[dict] = []  # 记录每次调用收到的参数，方便断言（Pydantic 会给每个实例复制一份默认值，不会共享）

    @property
    def _llm_type(self) -> str:
        return "fake"

    def _generate(self, messages, stop=None, run_manager=None, **kwargs) -> ChatResult:
        # kwargs 就是 llm.invoke(messages, temperature=0.3) 里额外传的参数
        self.calls.append({"messages": messages, **kwargs})
        reply = self.replies.pop(0)
        if isinstance(reply, Exception):
            raise reply
        return ChatResult(generations=[ChatGeneration(message=AIMessage(content=reply))])


def good_plan_json(days: int = 2) -> str:
    """一份合格的行程 JSON，景点和餐厅都来自 POI_RESPONSE"""

        plan_days.append({

            "attractions": [{"name": "故宫博物院", "address": "", "location": {"longitude": 1, "latitude": 1},
                             "visit_duration": 120, "description": "", "ticket_price": 999}],
            "meals": [
                {"type": "breakfast", "name": "酒店早餐", "estimated_cost": 0},
                {"type": "lunch", "name": "国家博物馆", "estimated_cost": 999},
                {"type": "dinner", "name": "故宫博物院", "estimated_cost": 999},
            ],
        })
    plan = {"city": "北京", "start_date": "2026-10-06", "end_date": f"2026-10-0{5 + days}", "days": plan_days,
```

> **LangChain 不是自带假模型吗？** `langchain_core` 里确实有 `GenericFakeChatModel`，可以按顺序返回预设的消息。但我们的测试还需要两样它不提供的东西：**记录每次调用的参数**（用来断言 temperature 逐次升高、重试时带上了错误原因），以及**在某一次调用时抛出异常**（用来模拟超时）。好在继承 `BaseChatModel` 只需要实现一个 `_generate` 方法，自己写一个也就 20 行。

```python
"""进度回调的类型定义（第 11 课）"""

from typing import Callable

# 进度回调：on_progress("正在搜索景点", 20)
# 规划器在关键步骤调用它，至于进度怎么展示（打印日志、推给前端……）由调用方决定
ProgressFn = Callable[[str, int], None]


def no_progress(stage: str, percent: int) -> None:
    """默认的回调：什么都不做。这样规划器内部不用到处判断 if on_progress is not None"""
```

```python
        "llm_configured": bool(settings.llm_api_key),
    }


def run_planner(request: TripRequest, planner: Planner | None, on_progress: ProgressFn = no_progress) -> TripPlanResponse:
    """普通接口和流式接口共用的核心逻辑（第 11 课从 plan_trip 里抽出来）"""
    if planner is None:
        plan = build_mock_plan(request)
        plan.budget = recompute_budget(plan, request)  # mock 数据也按统一口径计算预算
        on_progress("完成", 100)
        return TripPlanResponse(success=True, message="未配置大模型，返回 mock 数据", data=plan)

    result: PlanResult = planner.plan(request, on_progress=on_progress)
    if result.source == "fallback":
        reason = result.errors[-1] if result.errors else "未知原因"
        message = f"智能规划失败，返回基础行程。原因：{reason}"
    else:
        message = f"旅行计划生成成功（第 {result.attempts} 次生成）"
    if result.warnings:
        message += "；注意：" + "；".join(result.warnings[:3])
    return TripPlanResponse(success=True, message=message, data=result.plan)


@router.post("/plan", response_model=TripPlanResponse, summary="生成旅行计划")
def plan_trip(request: TripRequest, planner: Planner | None = Depends(get_trip_planner)):
    return run_planner(request, planner)


# ---------- 第 11 课：SSE 流式接口，推送真实进度 ----------
# SSE（Server-Sent Events）：服务端通过一个一直不关闭的 HTTP 响应持续推送消息，格式是
#   data: {...JSON...}\n\n
# 比 WebSocket 简单：单向推送（服务端 → 浏览器）、基于普通 HTTP、经过代理也没问题
@router.post("/plan/stream", summary="生成旅行计划（SSE 推送进度）")
def plan_trip_stream(request: TripRequest, planner: Planner | None = Depends(get_trip_planner)):
    # 线程安全的队列：后台线程往里放事件，响应生成器从里面取出来发给浏览器
    # 相当于一个"管道"，两头各在一个线程里
    events: queue.Queue[dict | None] = queue.Queue()

    def report(stage: str, percent: int) -> None:
        events.put({"type": "progress", "stage": stage, "percent": percent})

    def worker() -> None:
        try:
            response = run_planner(request, planner, on_progress=report)
            # mode="json"：保证所有字段都能被 json.dumps 序列化
            events.put({"type": "result", "data": response.model_dump(mode="json")})
        except Exception as exc:  # 高德 Key 错误等异常，也要通知前端，否则前端会一直等下去
            events.put({"type": "error", "message": str(exc)})
        finally:
            events.put(None)  # None 表示"结束了"

    # daemon=True：服务关闭时不用等这个线程结束
    threading.Thread(target=worker, daemon=True).start()

    def stream() -> Iterator[str]:
        # 海象运算符 :=（Python 3.8+）：赋值的同时判断，相当于
        # while (true) { const event = events.get(); if (event === null) break; ... }
        while (event := events.get()) is not None:
            yield f"data: {json.dumps(event, ensure_ascii=False)}\n\n"

    return StreamingResponse(
        stream(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",  # 不要缓存
            "X-Accel-Buffering": "no",  # 部署在 Nginx 后面时，告诉 Nginx 不要缓冲，否则进度会攒到最后一起发
        },
    )
```

### 为什么要用"线程 + 队列"？

```
请求线程（StreamingResponse 的生成器）         后台线程（worker）
        │                                         │
        │  events.get() 阻塞等待 ◀─── put(进度) ──┤ planner.plan(on_progress=report)
        │  yield "data: ...\n\n" → 浏览器          │   ├─ 收集数据   → report(10)
        │  events.get() 阻塞等待 ◀─── put(进度) ──┤   ├─ 调大模型   → report(50)
        │  ...                                    │   └─ 完成       → report(100)
        │  events.get() ◀──────────── put(result) ┤
        │  events.get() ◀──────────── put(None) ──┘ 结束
        ▼ 响应结束
```

规划过程是同步阻塞的，它没办法一边执行一边 `yield`。所以把它放到后台线程里执行，进度通过队列"递"给响应生成器。

```ts
// FastAPI 422 校验错误的单条结构：{ loc: ['body', 'party', 'adults'], msg: '...' }
interface ValidationErrorItem {
  loc: (string | number)[]
  msg: string
}

// 把各种错误统一转成一句人能看懂的中文
function toErrorMessage(error: unknown): string {
  if (!axios.isAxiosError(error)) return String(error)
  const err = error as AxiosError<{ detail?: string | ValidationErrorItem[] }>

  if (err.code === 'ECONNABORTED') return '请求超时，请稍后重试'
  // 没有 response = 请求根本没到后端（后端没启动、代理地址写错……）
  if (!err.response) return '无法连接后端服务，请确认后端已启动'

  const detail = err.response.data?.detail
  // 422：detail 是数组；去掉 loc 第一项 'body'，拼成 "party.adults: Input should be ..."
  if (Array.isArray(detail)) {
    return detail.map((d) => `${d.loc.slice(1).join('.') || '请求'}: ${d.msg}`).join('；')
  }
  // HTTPException：detail 是字符串
  return detail || `请求失败（HTTP ${err.response.status}）`
}

// 响应拦截器：失败时统一转换错误信息，业务代码只需要 catch 后显示 error.message
http.interceptors.response.use(
  (response) => response,
  (error) => Promise.reject(new Error(toErrorMessage(error))),
)

/** 生成旅行计划：成功返回行程和后端的提示信息，失败抛出 Error */
export async function generateTripPlan(request: TripRequest): Promise<{ plan: TripPlan; message: string }> {
  const { data } = await http.post<TripPlanResponse>('/api/trip/plan', request)
  // HTTP 200 但业务失败（success=false）也当成错误抛出
  if (!data.success || !data.data) {
    throw new Error(data.message || '生成失败')
  }
  // 第 10 课：把 message 一起返回。兜底方案、grounding 警告都写在 message 里（第 7、9 课）
  return { plan: data.data, message: data.message }
}

// ============================================================
// 第 11 课：SSE 流式接口，接收后端推送的真实进度
// ============================================================

type StreamEvent =
  | { type: 'progress'; stage: string; percent: number }
  | { type: 'result'; data: TripPlanResponse }
  | { type: 'error'; message: string }

/**
 * 为什么不用浏览器自带的 EventSource？它只支持 GET 请求，而我们要 POST 一个请求体
 * 所以用 fetch 拿到响应流，自己按 SSE 格式解析：每条消息是 "data: {...}\n\n"
 */
export async function generateTripPlanStream(
  request: TripRequest,
  onProgress: (stage: string, percent: number) => void,
): Promise<{ plan: TripPlan; message: string }> {
  const response = await fetch(`${baseURL}/api/trip/plan/stream`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(request),
  })
  if (!response.ok || !response.body) {
    throw new Error(`请求失败（HTTP ${response.status}）`)
  }

  const reader = response.body.getReader()
  const decoder = new TextDecoder() // 把字节流解码成字符串（中文是多字节字符，必须用 stream 模式解码）
  let buffer = '' // 网络分块不保证正好按消息边界切开，没处理完的部分先存着

  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })

    // 按空行切分出完整的消息，最后一段可能不完整，留到下一轮再处理
    const parts = buffer.split('\n\n')
    buffer = parts.pop() ?? ''

    for (const part of parts) {
      if (!part.startsWith('data: ')) continue
      const event = JSON.parse(part.slice('data: '.length)) as StreamEvent
      if (event.type === 'progress') {
        onProgress(event.stage, event.percent)
      } else if (event.type === 'error') {
        throw new Error(event.message)
      } else if (event.data.success && event.data.data) {
        return { plan: event.data.data, message: event.data.message }
      } else {
        throw new Error(event.data.message || '生成失败')
      }
    }
  }
  throw new Error('连接意外中断，没有收到结果')
}
```

### 1. 从 OpenAPI 自动生成 TS 类型（解决第 4 课"两边手动同步类型"的问题）

FastAPI 会自动生成 `/openapi.json`（就是 `/docs` 页面背后的数据），可以直接用它生成前端类型，不用再照着 `schemas.py` 手写。

**① 安装 + 加脚本**

```bash
cd frontend
npm i -D openapi-typescript
```

> 坑：openapi-typescript 7.x 的 peer 依赖写的是 `typescript@^5.x`，我们的项目用的是 TypeScript 6，直接安装会报 `ERESOLVE`。实测生成没有问题，所以在 `frontend/.npmrc` 里加一行，让 npm 不检查 peer 依赖的版本（这个文件要提交进 git，否则别人 `npm install` 也会报错）：
>
> ```ini
> legacy-peer-deps=true
> ```

`package.json` 的 `scripts` 里加：

```json
"gen:api": "openapi-typescript http://localhost:7001/openapi.json -o src/types/api.d.ts"
```

后端运行时执行 `npm run gen:api`，就会生成 `src/types/api.d.ts`。这个文件开头写着 "Do not make direct changes"，**不要手改**，下次生成会被覆盖。

**② 生成出来的结构**

```ts
export interface components {
  schemas: {
    TripRequest: { city: string; preferences?: string[]; /* ... */ }
    Budget: { total_attractions: number; /* ... */ readonly total: number }
    Meal: { type: "breakfast" | "lunch" | "dinner" | "snack"; /* ... */ }
    // 每个 Pydantic 模型一项
  }
}
export interface paths { /* 每个接口的请求、响应 */ }
```

几个对照点：

- `computed_field`（例如 `Budget.total`）会变成 `readonly`；
- `Literal[...]` 会变成字符串联合类型；
- 泛型 `ApiResponse[list[POIInfo]]` 会被展开成具体的名字 `ApiResponse_list_POIInfo__`，TS 里的泛型不会保留。

**③ `types/index.ts` 改成给生成的类型起别名**

组件和 `services/api.ts` 仍然从 `@/types` 导入，`index.ts` 继续作为唯一的类型入口，所以这些文件的 import 一行都不用改：

```ts
// 前端类型的统一入口：组件和 services 都从 '@/types' 导入，不直接碰 api.d.ts
// api.d.ts 由 `npm run gen:api` 从后端 /openapi.json 自动生成，不要手改
// 后端改了模型 → 启动后端 → npm run gen:api → npm run build，vue-tsc 会标出所有受影响的地方
import type { components } from './api'

type Schemas = components['schemas']

// Pydantic 里有默认值的字段，OpenAPI 会标成"可选"，生成的类型就带了 ?
// 对请求来说没错（前端可以不传，后端补默认值）；但响应里后端一定会输出这些字段
// 所以响应类型用 DeepRequired 递归去掉 ?（-? 只去掉"可能缺失"，Hotel | null 里的 null 会保留）
type DeepRequired<T> = T extends (infer U)[]
  ? DeepRequired<U>[]
  : T extends object
    ? { [K in keyof T]-?: DeepRequired<T[K]> }
    : T
type Resp<K extends keyof Schemas> = DeepRequired<Schemas[K]>

// ============ 请求 ============

export type TripRequest = Schemas['TripRequest']
export type PartyInfo = Schemas['PartyInfo']
export type BudgetConstraint = Schemas['BudgetConstraint']
// 从字段上取类型：后端 Literal['hard', 'soft', 'none'] → 'hard' | 'soft' | 'none'
export type BudgetStrictness = BudgetConstraint['strictness']

// ============ 响应（都套一层 Resp，去掉多余的 ?）============

export type Location = Resp<'Location'>
export type Attraction = Resp<'Attraction'>
export type Meal = Resp<'Meal'>
export type MealType = Meal['type']
export type Hotel = Resp<'Hotel'>
export type DayPlan = Resp<'DayPlan'>
export type WeatherInfo = Resp<'WeatherInfo'>
export type Budget = Resp<'Budget'>
export type TripPlan = Resp<'TripPlan'>
export type TripPlanResponse = Resp<'TripPlanResponse'>
export type POIInfo = Resp<'POIInfo'>
```

**④ 为什么需要 `DeepRequired`：有默认值的字段会变成可选**

刚换成生成的类型时，`vue-tsc` 一下报了 14 个错，全是 `'day.attractions' is possibly 'undefined'` 这种。原因是：

| 后端 | OpenAPI | 生成的 TS |
|---|---|---|
| `attractions: list[Attraction] = Field(default_factory=list)` | 不在 `required` 里 | `attractions?: Attraction[]` |
| `hotel: Hotel \| None = None` | 不在 `required` 里 | `hotel?: Hotel \| null` |

OpenAPI 的 `required` 表达的是"**请求时**能不能不传"。有默认值就能不传，所以标成可选。但**响应**里 Pydantic 会把所有字段都序列化出来（我们没开 `exclude_none`），这些字段一定存在。

所以分两种处理：

- **请求类型**（`TripRequest` 等）保持原样：`preferences?`、`budget_constraint?` 本来就可以不传；
- **响应类型**套一层 `Resp<...>`，递归去掉 `?`，和后端实际返回的数据保持一致。

> 另一种做法是从后端解决：给响应模型加 `model_config = ConfigDict(json_schema_serialization_defaults_required=True)`，让 Pydantic 在生成响应的 schema 时，把有默认值的字段也算作必填，前端就不需要 `DeepRequired` 了。这是更"正"的修法，留作作业 4。

**⑤ 剩下要手改的地方**

去掉多余的 `?` 后，只剩 1 处报错，而且这一处是真实存在的问题：结果页读取的 `request` 是 `TripRequest`，而它的 `budget_constraint` 本来就可以不传，所以要加 `?.`：

```vue
<!-- views/Result.vue -->
<p v-if="request?.budget_constraint?.amount" style="margin-top: 12px">
```

这正是自动生成类型的价值：手写类型时 `budget_constraint` 被写成了必填，这个问题就被藏住了。

**⑥ 以后的工作流**

后端改了模型 → 启动后端 → `npm run gen:api` → `npm run build`。`vue-tsc` 会把所有受影响的地方标出来，逐个修掉即可。

> 不在 OpenAPI 里的类型仍然手写，例如 `services/api.ts` 里 SSE 的 `StreamEvent`：流式推送的事件内容，OpenAPI 描述不了。

### 2. 修复参考仓库的 async 坑

参考仓库的 `api/routes/trip.py` 写的是 `async def plan_trip`，里面调用的却是同步的 `agent.plan_trip()`，**这会阻塞整个事件循环**：一个用户在生成行程的那一两分钟里，其他所有请求（包括 `/health`）都会卡住。我们从第 5 课起就用 `def`，避开了这个问题（原理见第 5 课的表格）。
