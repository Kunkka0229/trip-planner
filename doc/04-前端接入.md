## Step 2：配置别名和代理

`vite.config.ts`：

```ts
import { fileURLToPath, URL } from 'node:url'
import vue from '@vitejs/plugin-vue'
import { defineConfig } from 'vite'

export default defineConfig({
  plugins: [vue()],
  resolve: {
    alias: {
      // 让 import '@/types' 指向 src/types，避免写一长串 ../../
      // （还需要在 tsconfig.app.json 里配 paths，TS 才认识这个别名）
      '@': fileURLToPath(new URL('./src', import.meta.url)),
    },
  },
  server: {
    port: 5173,
    // 开发代理：浏览器请求 http://localhost:5173/api/xxx
    // → Vite 开发服务器转发到 http://127.0.0.1:7001/api/xxx
    // 对浏览器来说请求的是同源地址，所以开发阶段根本不会触发 CORS
    proxy: {
      '/api': {
        target: 'http://127.0.0.1:7001',
        changeOrigin: true, // 把请求头里的 Host 改成目标地址
      },
    },
  },
})
```

`tsconfig.app.json` 的 `compilerOptions` 里追加（Vite 负责打包时的别名解析，TS 负责类型检查时的别名解析，两边都要配）：

```jsonc
    /* 路径别名 */
    "paths": {
      "@/*": ["./src/*"]
    }
```

`.env.example`（然后 `cp .env.example .env`）：

```bash
# 后端地址：开发环境留空，走 Vite 的 /api 代理；
# 生产环境前后端不同域名时，填后端完整地址，例如 https://api.example.com
VITE_API_BASE_URL=
```

### 代理 vs CORS：第 2 课配的 CORS 还有用吗？

| 场景 | 请求路径 | 需要 CORS 吗 |
|---|---|---|
| 开发，走 Vite 代理（本课用法） | 浏览器 → `localhost:5173/api` → Vite 转发 → `127.0.0.1:7001` | **不需要**：浏览器只和 5173 打交道，是同源请求 |
| 开发，`VITE_API_BASE_URL=http://127.0.0.1:7001` | 浏览器 → `127.0.0.1:7001` | **需要**：端口不同就算跨域 |
| 生产，前后端不同域名 | 浏览器 → `api.example.com` | **需要** |
| 生产，Nginx 把 `/api` 反向代理到后端 | 和开发代理一样 | 不需要 |

所以第 2 课的 CORS 配置并没有白做：等你部署时前后端不在同一个域名，就会用上。

```ts
// 后端改了模型，这里也要同步改（第 11 课改成从 OpenAPI 自动生成，就不用手动同步了）

  children: number

  city: string
  start_date: string // 'YYYY-MM-DD'，JSON 里没有日期类型，统一用字符串
  end_date: string
  travel_days: number
  transportation: string
  accommodation: string
  preferences: string[]
  free_text_input: string
  party: PartyInfo
  budget_constraint: BudgetConstraint

// ============ 响应 ============

  longitude: number
  latitude: number
}

export interface Attraction {
  name: string
  address: string
  location: Location
  visit_duration: number // 分钟
  description: string
  ticket_price: number // 元/人
}

export type MealType = 'breakfast' | 'lunch' | 'dinner' | 'snack'

export interface Meal {
  type: MealType
  name: string
  estimated_cost: number // 元/人
}

export interface Hotel {
  name: string
  address: string
  location: Location | null // 后端是 Location | None，序列化后是 null，不是 undefined
  estimated_cost: number // 元/间/晚
}

export interface DayPlan {
  date: string
  day_index: number
  description: string
  hotel: Hotel | null
  attractions: Attraction[]
  meals: Meal[]
}

export interface WeatherInfo {
  date: string
  day_weather: string
  day_temp: number | string // 远期天气可能是 "未知"
  night_temp: number | string
}

export interface Budget {
  total_attractions: number
  total_hotels: number
  total_meals: number
  total_transportation: number
  total: number // 响应里有 total（后端 computed_field 输出）
}

export interface TripPlan {
  city: string
  start_date: string
  end_date: string
  days: DayPlan[]
  weather_info: WeatherInfo[]
  overall_suggestions: string
  budget: Budget | null
}

// 统一响应外壳，用泛型复用：ApiResponse<TripPlan>、ApiResponse<POI[]>……
export interface ApiResponse<T> {
  success: boolean
  message: string
  data: T | null
}

export type TripPlanResponse = ApiResponse<TripPlan>
```

> **Python ↔ TS 类型对照**：`str` → `string`，`int`/`float` → `number`，`X | None` → `X | null`，`list[X]` → `X[]`，`Literal['a','b']` → `'a' | 'b'`，`date` → `string`。
>
> 注意可选值要写 **`| null`**，不要写成 `?:`：Pydantic 序列化 `None` 时会输出 `null`，字段本身不会缺失。

## Step 4：API 服务层 `src/services/api.ts`

```ts
import axios, { AxiosError } from 'axios'
import type { TripPlan, TripPlanResponse, TripRequest } from '@/types'

// 读取 .env 里的 VITE_API_BASE_URL（只有 VITE_ 开头的变量才会暴露给前端代码）
// 留空 → baseURL 为 ''，请求发到当前域名的 /api/...，由 Vite 代理转发
const baseURL = (import.meta.env.VITE_API_BASE_URL ?? '').trim().replace(/\/$/, '')

const http = axios.create({
  baseURL,
  // 大模型生成行程可能要 1~3 分钟，axios 默认不超时，这里给 5 分钟上限
  timeout: 5 * 60 * 1000,
  headers: { 'Content-Type': 'application/json' },
})

// FastAPI 422 校验错误的单条结构：{ loc: ['body', 'party', 'adults'], msg: '...' }
interface ValidationErrorItem {
  loc: (string | number)[]
  msg: string
}

// 把各种错误统一转成一句人能看懂的中文
function toErrorMessage(error: unknown): string {
  if (!axios.isAxiosError(error)) return String(error)
  const err = error as AxiosError<{ detail?: string | ValidationErrorItem[] }>

  if (err.code === 'ECONNABORTED') return '请求超时，请稍后重试'
  // 没有 response = 请求根本没到后端（后端没启动、代理地址写错……）
  if (!err.response) return '无法连接后端服务，请确认后端已启动'

  const detail = err.response.data?.detail
  // 422：detail 是数组；去掉 loc 第一项 'body'，拼成 "party.adults: Input should be ..."
  if (Array.isArray(detail)) {
    return detail.map((d) => `${d.loc.slice(1).join('.') || '请求'}: ${d.msg}`).join('；')
  }
  // HTTPException：detail 是字符串
  return detail || `请求失败（HTTP ${err.response.status}）`
}

// 响应拦截器：失败时统一转换错误信息，业务代码只需要 catch 后显示 error.message
http.interceptors.response.use(
  (response) => response,
  (error) => Promise.reject(new Error(toErrorMessage(error))),
)

/** 生成旅行计划：成功直接返回 TripPlan，失败抛出 Error */
export async function generateTripPlan(request: TripRequest): Promise<TripPlan> {
  const { data } = await http.post<TripPlanResponse>('/api/trip/plan', request)
  // HTTP 200 但业务失败（success=false）也当成错误抛出
  if (!data.success || !data.data) {
    throw new Error(data.message || '生成失败')
  }
  return data.data
}
```

```vue
<script setup lang="ts">
import { computed, reactive, ref } from 'vue'
import { useRouter } from 'vue-router'
import { message } from 'ant-design-vue'
import type { Dayjs } from 'dayjs'
import { generateTripPlan } from '@/services/api'
import type { BudgetStrictness, TripRequest } from '@/types'

const router = useRouter()

// 表单状态和接口请求体分开：
// 表单里的日期是 Dayjs 对象（日期组件需要），提交时再转成 'YYYY-MM-DD' 字符串
const form = reactive({
  city: '',
  dates: null as [Dayjs, Dayjs] | null, // a-range-picker 的值：[开始, 结束]
  transportation: '公共交通',
  accommodation: '经济型酒店',
  preferences: [] as string[],
  freeText: '',
  adults: 2,
  children: 0,
  elders: 0,
  budgetAmount: null as number | null, // null = 不限预算
  strictness: 'soft' as BudgetStrictness,
})

const transportationOptions = ['公共交通', '打车/网约车', '自驾', '骑行/步行']
const accommodationOptions = ['经济型酒店', '舒适型酒店', '高端酒店', '民宿']
const preferenceOptions = ['历史文化', '自然风光', '美食', '购物', '艺术', '亲子友好']

// 天数由日期计算得出，用户不用填，也就不会和日期对不上（后端 model_validator 会校验）
const travelDays = computed(() => {
  if (!form.dates) return 0
  const [start, end] = form.dates
  return end.diff(start, 'day') + 1
})

// ---------- 假进度条（书 13.6.2） ----------
// 后端只有一个同步接口，前端拿不到真实进度，只能"演"一个进度条安抚用户
// 第 11 课会用 SSE 改成后端推送的真实进度
const loading = ref(false)
const progress = ref(0)
const progressText = ref('')

function startFakeProgress(): number {
  progress.value = 0
  const steps = ['正在搜索景点...', '正在查询天气...', '正在推荐酒店...', '正在生成行程...']
  // 每 0.5 秒涨 10%，最多涨到 90%，剩下 10% 等真正返回后再补满
  return window.setInterval(() => {
    if (progress.value < 90) {
      progress.value += 10
      progressText.value = steps[Math.min(Math.floor(progress.value / 25), steps.length - 1)]
    }
  }, 500)
}

// a-form 校验通过后才会触发 @finish
async function handleSubmit() {
  if (!form.dates) {
    message.error('请选择出行日期')
    return
  }
  const [start, end] = form.dates

  // 组装成和后端 TripRequest 完全一致的结构
  const request: TripRequest = {
    city: form.city.trim(),
    start_date: start.format('YYYY-MM-DD'),
    end_date: end.format('YYYY-MM-DD'),
    travel_days: travelDays.value,
    transportation: form.transportation,
    accommodation: form.accommodation,
    preferences: form.preferences,
    free_text_input: form.freeText,
    party: { adults: form.adults, children: form.children, elders: form.elders },
    budget_constraint: {
      amount: form.budgetAmount,
      // 没填预算时，严格程度没有意义，固定为 none
      strictness: form.budgetAmount === null ? 'none' : form.strictness,
    },
  }

  loading.value = true
  const timer = startFakeProgress()
  try {
    const plan = await generateTripPlan(request)
    progress.value = 100
    // 用 sessionStorage 把结果带到结果页：刷新结果页数据还在，关掉标签页就清空
    sessionStorage.setItem('tripPlan', JSON.stringify(plan))
    message.success('行程生成成功')
    router.push({ name: 'result' })
  } catch (error) {
    // api.ts 的拦截器已经把错误转成了可读的中文
    message.error((error as Error).message)
  } finally {
    clearInterval(timer) // 无论成功失败都要清掉定时器，否则会一直跑
    loading.value = false
  }
}
</script>
```
