## Step 2：工具函数

`src/utils/budget.ts`：

```ts
import type { Budget, TripPlan } from '@/types'

/**
 * 前端重算预算：编辑行程（删除、调整景点）之后，预算要跟着变
 * 口径必须和后端 app/agents/output.py 的 recompute_budget 完全一致：
 *   门票、餐饮 × 人数；酒店 × 房间数（2 人一间，向上取整）；交通沿用原值
 * 这是"同一套逻辑写了两遍"，作业 2 会讨论更好的做法
 */
export function recomputeBudget(plan: TripPlan, people: number): Budget {
  const rooms = Math.ceil(people / 2)
  const sum = (values: number[]) => values.reduce((a, b) => a + b, 0)

  const total_attractions = sum(plan.days.flatMap((d) => d.attractions.map((a) => a.ticket_price))) * people
  const total_meals = sum(plan.days.flatMap((d) => d.meals.map((m) => m.estimated_cost))) * people
  const total_hotels = sum(plan.days.map((d) => d.hotel?.estimated_cost ?? 0)) * rooms
  const total_transportation = plan.budget?.total_transportation ?? 0

  return {
    total_attractions,
    total_meals,
    total_hotels,
    total_transportation,
    total: total_attractions + total_meals + total_hotels + total_transportation,
  }
}
```

```ts
import html2canvas from 'html2canvas'
import { jsPDF } from 'jspdf'

/** 把一个 DOM 元素"截图"成 canvas（html2canvas 会把 DOM 和 CSS 重新绘制到 canvas 上，并不是真正的截屏） */
async function renderToCanvas(element: HTMLElement): Promise<HTMLCanvasElement> {
  return html2canvas(element, {
    scale: 2, // 2 倍分辨率，导出的图片在高清屏上不会模糊
    useCORS: true, // 允许加载跨域图片（需要图片服务器也允许跨域）
    backgroundColor: '#f6f8fb',
    // 忽略带 data-export-ignore 属性的元素（比如编辑按钮、导出按钮本身）
    ignoreElements: (el) => el.hasAttribute('data-export-ignore'),
  })
}

/** 导出为 PNG 图片 */
export async function exportAsImage(element: HTMLElement, filename: string): Promise<void> {
  const canvas = await renderToCanvas(element)
  // 创建一个临时的 <a download> 并模拟点击，触发浏览器下载
  const link = document.createElement('a')
  link.download = `${filename}.png`
  link.href = canvas.toDataURL('image/png')
  link.click()
}

/** 导出为 PDF：把长图按 A4 页高切成多页 */
export async function exportAsPdf(element: HTMLElement, filename: string): Promise<void> {
  const canvas = await renderToCanvas(element)
  const pdf = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' })

  const pageWidth = pdf.internal.pageSize.getWidth() // 210mm
  const pageHeight = pdf.internal.pageSize.getHeight() // 297mm
  // 图片宽度撑满页面，按比例算出整张长图在 PDF 里的高度
  const imgHeight = (canvas.height * pageWidth) / canvas.width
  const imgData = canvas.toDataURL('image/jpeg', 0.92) // JPEG 比 PNG 小很多

  // 同一张长图贴到每一页，每页向上偏移一个页高，相当于"分页滚动"
  let offset = 0
  while (offset < imgHeight) {
    if (offset > 0) pdf.addPage()
    pdf.addImage(imgData, 'JPEG', 0, -offset, pageWidth, imgHeight)
    offset += pageHeight
  }
  pdf.save(`${filename}.pdf`)
}
```

## Step 3：地图组件 `src/components/TripMap.vue`

```vue
<script setup lang="ts">
import { onMounted, onUnmounted, ref, shallowRef, watch } from 'vue'
import AMapLoader from '@amap/amap-jsapi-loader'
import type { DayPlan } from '@/types'

const props = defineProps<{
  days: DayPlan[]
  activeDay: number // -1 = 显示全部天数，0/1/2… = 只显示第几天
}>()

// 每天一种颜色，标记点和路线都用这个颜色
const DAY_COLORS = ['#1677ff', '#fa541c', '#13c2c2', '#722ed1', '#52c41a', '#eb2f96', '#faad14']

const container = ref<HTMLDivElement>()
const error = ref('')
// 高德的 AMap 对象和地图实例没有 TS 类型，用 any
// shallowRef：只追踪 .value 本身的替换，不把整个地图对象变成深层响应式（否则会非常卡）
const AMap = shallowRef<any>(null)
const map = shallowRef<any>(null)

/** 根据当前数据重新绘制标记点和路线 */
function draw() {
  if (!map.value || !AMap.value) return
  map.value.clearMap() // 清掉上一次画的覆盖物

  props.days.forEach((day, dayIndex) => {
    if (props.activeDay !== -1 && props.activeDay !== dayIndex) return
    const color = DAY_COLORS[dayIndex % DAY_COLORS.length]
    // 高德要求坐标格式是 [经度, 纬度]
    const path = day.attractions.map((a) => [a.location.longitude, a.location.latitude])

    day.attractions.forEach((attraction, i) => {
      const marker = new AMap.value.Marker({
        position: path[i],
        title: attraction.name,
        // 用 HTML 自定义标记的样式：圆点里显示"天数-序号"
        content: `<div style="background:${color};color:#fff;border-radius:12px;padding:2px 8px;font-size:12px;white-space:nowrap">${dayIndex + 1}-${i + 1} ${attraction.name}</div>`,
        offset: new AMap.value.Pixel(-12, -12),
      })
      map.value.add(marker)
    })

    // 至少两个点才能连成线
    if (path.length > 1) {
      map.value.add(new AMap.value.Polyline({ path, strokeColor: color, strokeWeight: 4, strokeOpacity: 0.8 }))
    }
  })

  map.value.setFitView() // 自动缩放，让所有覆盖物都能显示在视野里
}

onMounted(async () => {
  const key = import.meta.env.VITE_AMAP_JS_KEY
  if (!key) {
    error.value = '未配置 VITE_AMAP_JS_KEY，地图不可用（其他功能不受影响）'
    return
  }
  // 安全密钥必须在加载 JS API 之前设置好
  if (import.meta.env.VITE_AMAP_SECURITY_CODE) {
    window._AMapSecurityConfig = { securityJsCode: import.meta.env.VITE_AMAP_SECURITY_CODE }
  }
  try {
    // 动态加载高德 JS API（相当于动态插入一个 <script src="https://webapi.amap.com/maps?...">）
    AMap.value = await AMapLoader.load({ key, version: '2.0' })
    map.value = new AMap.value.Map(container.value, { zoom: 12, viewMode: '2D' })
    draw()
  } catch (e) {
    error.value = `地图加载失败: ${String(e)}`
  }
})

// 编辑行程、切换天数时重新绘制；deep: true 才能监听到数组内部的变化（比如调整了景点顺序）
watch(() => [props.days, props.activeDay], draw, { deep: true })

// 组件卸载时销毁地图，释放 WebGL 资源，否则反复进出页面会内存泄漏
onUnmounted(() => map.value?.destroy())
</script>

<template>
  <a-alert v-if="error" :message="error" type="warning" show-icon />
  <div v-else ref="container" class="trip-map" />
</template>

<style scoped>
.trip-map {
  width: 100%;
  height: 420px;
  border-radius: 8px;
}
</style>
```

> **为什么地图对象要用 `shallowRef`？** 用 `ref` 的话，Vue 会把地图对象内部成千上万个属性都转成响应式 Proxy，页面会明显卡顿，甚至可能报错。像地图实例、图表实例、编辑器实例这类第三方大对象，都应该用 `shallowRef`，或者用普通变量存放。

### 已知限制：导出的图片里地图是空白的

高德地图用 WebGL 绘制，html2canvas 无法读取 WebGL 画布的内容。参考仓库的做法是：导出前先调用地图 canvas 的 `toDataURL()` 生成快照，替换掉地图容器，再执行导出（见参考仓库 `Result.vue` 里的 `replaceMapSnapshots`）。这留作选做作业 4。
