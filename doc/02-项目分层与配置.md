# 第 2 课：项目分层与配置管理

**目标**：把第 1 课"全写在一个文件里"的代码，拆成和参考仓库一致的结构。学会三件事：

1. **APIRouter**：按业务拆分路由，类似 Express 的 `express.Router()`。
2. **pydantic-settings**：从 `.env` 读配置，并自动做类型转换，类似 Vite 的 `import.meta.env`，但更强。
3. **CORS 中间件 + lifespan**：让前端能跨域访问，并在启动时执行初始化逻辑。

## Step 3：配置中心 `app/config.py`

```python
"""应用配置：从环境变量和 backend/.env 文件读取"""

from functools import lru_cache  # 标准库里的缓存装饰器，这里用来实现"单例"
from pathlib import Path         # 处理文件路径的标准库，类似 Node 的 path 模块

from pydantic_settings import BaseSettings, SettingsConfigDict

# backend/ 目录的绝对路径
# __file__              = .../backend/app/config.py（当前文件）
# .resolve()            = 转成绝对路径
# .parent               = .../backend/app
# .parent.parent        = .../backend
# 为什么不直接写 ".env"？相对路径是相对于"你在哪个目录执行命令"（cwd），
# 在别的目录启动时就会找不到 .env。用 __file__ 算出来的路径永远准确。
BACKEND_DIR = Path(__file__).resolve().parent.parent


class Settings(BaseSettings):
    """
    继承 BaseSettings 后，每个字段都会自动去找同名的环境变量（不区分大小写）：
      字段 amap_api_key  ←→  环境变量 AMAP_API_KEY
    读取优先级（高 → 低）：
      1. 真实的环境变量（比如 PORT=8001 uv run python run.py）
      2. .env 文件
      3. 这里写的默认值
    """

    # model_config：这个配置类本身的行为设置
    model_config = SettingsConfigDict(
        env_file=BACKEND_DIR / ".env",  # 从哪个文件读
        env_file_encoding="utf-8",      # .env 里可能有中文
        extra="ignore",                 # .env 里有未声明的变量时忽略，不报错
    )

    # ---------- 应用信息 ----------
    app_name: str = "HelloAgents 智能旅行助手"
    app_version: str = "0.1.0"

    # ---------- 服务器 ----------
    host: str = "0.0.0.0"  # 0.0.0.0 = 监听所有网卡，局域网其他设备也能访问
    port: int = 7001       # 声明成 int：.env 里写的是字符串 "7001"，会被自动转成整数
                           # 如果写成 PORT=abc，启动时直接报错，而不是运行到一半才出问题

    # 允许跨域的前端地址，多个用英文逗号分隔
    # 为什么用 str 而不是 list[str]？pydantic-settings 对 list 类型要求环境变量写成
    # JSON 格式（["http://a","http://b"]），在 .env 里写起来很别扭，所以存字符串再自己拆
    cors_origins: str = "http://localhost:5173,http://127.0.0.1:5173"

    # ---------- 第三方服务密钥（后面的课会用到，先占位） ----------
    amap_api_key: str = ""   # 高德地图 Web 服务 Key
    llm_api_key: str = ""    # 大模型 API Key
    llm_base_url: str = ""   # 大模型接口地址（OpenAI 兼容格式）
    llm_model_id: str = ""   # 模型名称

    log_level: str = "INFO"

    # @property：把方法变成"只读属性"，调用时不用加括号：settings.cors_origins_list
    # 类似 JS class 里的 get corsOriginsList() { ... }
    @property
    def cors_origins_list(self) -> list[str]:
        """把逗号分隔的字符串拆成列表，并去掉空白和空项"""
        # 列表推导式，相当于 JS 的：
        # this.cors_origins.split(",").map(o => o.trim()).filter(o => o)
        return [o.strip() for o in self.cors_origins.split(",") if o.strip()]


# @lru_cache：缓存函数返回值。第一次调用时创建 Settings 并读取 .env，
# 之后每次调用都直接返回同一个对象，相当于单例，不会重复读文件
@lru_cache
def get_settings() -> Settings:
    return Settings()
```

## Step 5：拆路由

### `app/api/routes/demo.py`（第 1 课的练习接口，搬过来）

```python
"""练习用接口：第 1 课写的东西都放这里"""

from fastapi import APIRouter

# APIRouter 相当于 express.Router()：一个"迷你 app"，最后挂到主 app 上
# prefix：这个文件里所有路由的公共前缀
# tags：在 /docs 页面里把这些接口归为一组显示
router = APIRouter(prefix="/demo", tags=["练习"])


# 注意这里用的是 @router.get，不是 @app.get
# 最终地址 = include_router 的 prefix(/api) + 这里的 prefix(/demo) + 路径(/hello)
#          = /api/demo/hello
@router.get("/hello")
def hello(name: str = "旅行者"):  # 第 1 课作业 1 的参考答案
    # f"..." 是 f-string，相当于 JS 的模板字符串 `你好，${name}`
    return {"greeting": f"你好，{name}"}


# 最终地址：/api/demo/cities/{city}
@router.get("/cities/{city}")
def get_city(city: str, days: int = 3):
    return {"city": city, "days": days}
```

### `app/api/routes/trip.py`（旅行规划业务）

```python
"""旅行规划相关接口"""

from fastapi import APIRouter

from app.config import get_settings  # 从配置中心取配置

router = APIRouter(prefix="/trip", tags=["旅行规划"])


# 最终地址：/api/trip/health
# 和根路径的 /health 不同：这里检查的是"旅行规划这个业务"依赖的配置是否齐全
@router.get("/health")
def trip_health():
    settings = get_settings()  # 有 lru_cache，拿到的永远是同一个对象
    return {
        "status": "healthy",
        "service": "trip-planner",
        # bool("") → False，bool("xxx") → True
        # 只返回"有没有配置"，绝对不要把密钥本身返回给前端
        "amap_configured": bool(settings.amap_api_key),
        "llm_configured": bool(settings.llm_api_key),
    }
```

## Step 6：主应用 `app/api/main.py`

把旧的 `app/main.py` **删掉**，新建 `app/api/main.py`：

```python
"""FastAPI 主应用：创建 app、挂中间件、注册路由"""

from contextlib import asynccontextmanager  # 用来写 lifespan 的工具

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware  # FastAPI 自带的 CORS 中间件

from app.api.routes import demo, trip  # 导入两个路由模块
from app.config import get_settings

settings = get_settings()


def status_of(value: str) -> str:
    """密钥只显示"已配置/未配置"，避免在日志里泄露"""
    return "已配置" if value else "未配置"


# ---------- lifespan：应用的生命周期 ----------
# yield 之前的代码：服务启动时执行一次（适合：校验配置、建数据库连接、预热模型）
# yield 之后的代码：服务关闭时执行一次（适合：关闭连接、释放资源）
# 类似 Vue 组件的 onMounted / onUnmounted，只不过作用对象是整个后端服务
# 参考仓库用的 @app.on_event("startup") 是旧写法，FastAPI 已经不推荐使用
@asynccontextmanager
async def lifespan(app: FastAPI):
    print("=" * 50)  # 字符串乘法：把 "=" 重复 50 次
    print(f"🚀 {settings.app_name} v{settings.app_version}")
    print(f"   监听地址: {settings.host}:{settings.port}")
    print(f"   高德 Key: {status_of(settings.amap_api_key)}")
    print(f"   LLM Key : {status_of(settings.llm_api_key)}")
    print(f"   CORS 允许: {settings.cors_origins_list}")
    print("=" * 50)
    yield  # ← 服务在这里运行，直到收到关闭信号
    print("👋 服务已关闭")


# 创建应用：标题和版本从配置读取，生命周期交给 lifespan
app = FastAPI(
    title=settings.app_name,
    version=settings.app_version,
    lifespan=lifespan,
)


# ---------- CORS 中间件 ----------
# 浏览器的同源策略：前端在 localhost:5173，后端在 localhost:7001，端口不同就算"跨域"，
# 浏览器会拦下响应。后端在响应头里加上 Access-Control-Allow-Origin，浏览器才会放行。
# 相当于 Express 里的 app.use(cors({ origin: [...] }))
app.add_middleware(
    CORSMiddleware,
    allow_origins=settings.cors_origins_list,  # 只允许这些前端地址
    allow_credentials=True,                    # 允许携带 cookie
    allow_methods=["*"],                       # 允许所有 HTTP 方法（GET/POST/...）
    allow_headers=["*"],                       # 允许所有请求头
)


# ---------- 注册路由 ----------
# 相当于 Express 的 app.use('/api', tripRouter)
# prefix="/api" 加上路由文件里自己的 prefix，组成完整路径
app.include_router(demo.router, prefix="/api")
app.include_router(trip.router, prefix="/api")


@app.get("/")
def root():
    """根路径：返回服务基本信息"""
    return {
        "name": settings.app_name,
        "version": settings.app_version,
        "docs": "/docs",
    }


@app.get("/health")
def health():
    """服务级健康检查：只要进程活着就返回 healthy"""
    return {"status": "healthy"}
```

## Step 7：启动脚本 `run.py`

```python
"""启动脚本：python run.py"""

import uvicorn  # ASGI 服务器，真正负责监听端口

from app.config import get_settings

# 只有直接执行 python run.py 时才会进入这个 if
# 被别的文件 import 时不会执行，类似 Node 里的 if (require.main === module)
if __name__ == "__main__":
    settings = get_settings()

    uvicorn.run(
        # 用字符串 "模块路径:变量名" 指定 app，而不是直接传 app 对象
        # 原因：reload=True 时，uvicorn 要在子进程里重新 import 你的代码，只能用字符串
        # app.api.main 对应文件 app/api/main.py，app 是里面的变量名
        "app.api.main:app",
        host=settings.host,
        port=settings.port,                       # 端口从配置读取
        reload=True,                              # 代码改动后自动重启（仅开发环境）
        log_level=settings.log_level.lower(),     # uvicorn 要求小写：info / debug
    )
```

## Step 8：启动并验证

```bash
# 方式一：用启动脚本，端口读取 .env 里的 PORT（7001）
uv run python run.py

# 方式二：fastapi dev 也能用，但它固定用 8000 端口，不会读取我们的配置
# uv run fastapi dev app/api/main.py
```

> **坑：为什么不用参考仓库的 7000 端口？**
> macOS 12 及以上，系统的"隔空播放接收器"（AirPlay，进程名 `ControlCenter`）默认占用 **5000 和 7000** 端口，用 7000 启动会报 `[Errno 48] Address already in use`。
> 更隐蔽的是：如果服务其实没启动成功，用 curl 请求 7000 端口时，会收到 AirPlay 返回的 `403 Forbidden`，响应头里是 `Server: AirTunes/...`。**遇到奇怪的响应，先看 `Server` 响应头，确认请求打到了谁身上。**
>
> 排查端口占用：
> ```bash
> # 查看谁在监听 7000 端口；COMMAND 是进程名，PID 是进程号
> lsof -nP -iTCP:7000 -sTCP:LISTEN
> ```
> 如果占用者是你自己忘了关的 `python` 进程，可以 `kill <PID>` 结束它；如果是系统进程，就换个端口。我们统一使用 **7001**。

打开 http://127.0.0.1:7001/docs，你会看到接口按 **练习** 和 **旅行规划** 两组显示（就是 `tags` 的效果）。

用 curl 验证（另开一个终端）：

```bash
# 1. 基础接口
curl http://127.0.0.1:7001/api/trip/health

# 2. 中文参数要用 --data-urlencode 编码，直接写 ?name=小明 会报 Invalid HTTP request
#    浏览器会自动编码，curl 不会。-G 表示把参数拼到 URL 上，用 GET 请求
curl -G --data-urlencode "name=小明" http://127.0.0.1:7001/api/demo/hello

# 3. 验证 CORS：模拟从前端页面发出的请求（-H 加请求头，-i 显示响应头）
curl -i -H "Origin: http://localhost:5173" http://127.0.0.1:7001/health
# 应该能看到：access-control-allow-origin: http://localhost:5173

# 4. 换一个不在白名单里的来源
curl -i -H "Origin: http://evil.com" http://127.0.0.1:7001/health
# 响应体照样返回，但响应头里没有 access-control-allow-origin
```

> **重要理解**：CORS 拦截的是**浏览器**，不是服务器。第 4 条请求，curl 依然拿到了数据，只是响应头里缺少允许跨域的标记；如果是浏览器发起的请求，它看到缺少这个头就会拦下响应，不交给你的 JS。所以 CORS **不是安全防护**，不能用它来保护接口。
